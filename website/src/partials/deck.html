Presentations

AI Agents Build Your Decks

Agents draft slides, arrange layouts, insert visuals, and polish your presentation. A full deck editor where every action is agent-driven.

slides.ts 
// Create a deck
const deck = await claw
  .slides.create({
    title: "Q2 Product Update",
    theme: "dark"
  });

// Add a slide
await claw.slides
  .addSlide(deck.id, {
    layout: "title",
    heading: "Growth"
  });

// Insert an image
await claw.slides
  .insert("slide_3", {
    type: "image",
    src: "chart.png"
  });
